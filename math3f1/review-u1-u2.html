خصائص المساواة

  • ما تفعله بطرف افعله بالآخر.
  • معامل كسري ⟸ اضرب في المقلوب.

حالات خاصة

  • عبارة صحيحة دائمًا ⟸ متطابقة
  • عبارة خاطئة ⟸ ∅

القيمة المطلقة

  • |أ| = ب ⟸ أ = ±ب
  • |أ| = سالب ⟸ ∅

ترجمة الجمل والتحقق من الحل

١-١

القاعدة

ترجم الكلمات إلى رموز: أمثال ⟸ ضرب، مضافًا إليه ⟸ +، مطروحًا منه ⟸ −. للتحقق: عوّض القيمة وتأكد من تساوي الطرفين.

مثال محلول بصيغة الاختبار

ما المعادلة التي تمثل الجملة: «ثلاثة أمثال عدد مضافًا إليه ٧ يساوي ٣١»؟
  1. أ) ٧ن + ٣ = ٣١
  2. ب) ٣ن + ٧ = ٣١
  3. ج) ٣(ن + ٧) = ٣١
  4. د) ٣ن = ٣١ + ٧
  1. «ثلاثة أمثال عدد» ⟸ ٣ن، و«مضافًا إليه ٧» ⟸ + ٧، و«يساوي» ⟸ =.
  2. المعادلة: ٣ن + ٧ = ٣١، وحلها ن = ٨.
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١أيّ القيم الآتية حلّ للمعادلة ك − ١ = ٢؟
الحل:
  1. نعوّض كل قيمة في المعادلة ونبحث عن القيمة التي تجعلها صحيحة.
  2. عند ك = ٣: ١(٣) − ١ = ٢ ✓
٢ما المعادلة التي تمثل الجملة: «ثلاثة أمثال عدد مضافًا إليه ٤ يساوي ٢٢»؟
الحل:
  1. «ثلاثة أمثال عدد» ⟸ ٣ن، و«مضافًا إليه ٤» ⟸ + ٤، و«يساوي» ⟸ =.
  2. المعادلة: ٣ن + ٤ = ٢٢، وحلها ن = ٦.
٣أيّ القيم الآتية حلّ للمعادلة −٣م − ٨ = ٧؟
الحل:
  1. نعوّض كل قيمة في المعادلة ونبحث عن القيمة التي تجعلها صحيحة.
  2. عند م = −٥: −٣(−٥) − ٨ = ٧ ✓

المعادلات ذات الخطوة الواحدة

١-٢

القاعدة

استعمل العملية العكسية للطرفين: الجمع ↔ الطرح، الضرب ↔ القسمة، ومع المعامل الكسري اضرب في المقلوب.

مثال محلول بصيغة الاختبار

ما حل المعادلة س + ٦ = ٣؟
  1. أ) س = ٩
  2. ب) س = ٣
  3. ج) س = −٣
  4. د) س = −٢
  1. نطرح ٦ من الطرفين.
  2. س = −٣، وللتحقق نعوّض الحل في المعادلة الأصلية.
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما حل المعادلة ٥٦م = ٥؟
الحل:
  1. نضرب الطرفين في مقلوب المعامل ٦٥.
  2. م = ٦، وللتحقق نعوّض الحل في المعادلة الأصلية.
٢ما حل المعادلة ن − ١٥ = −٤؟
الحل:
  1. نضيف ١٥ إلى الطرفين.
  2. ن = ١١، وللتحقق نعوّض الحل في المعادلة الأصلية.
٣ما حل المعادلة ٥ك = −٦٠؟
الحل:
  1. نقسم الطرفين على ٥.
  2. ك = −١٢، وللتحقق نعوّض الحل في المعادلة الأصلية.

المعادلات المتعددة الخطوات

١-٣

القاعدة

تخلّص من الحد الثابت أولًا ثم من المعامل. مع س + أب اضرب في ب أولًا.

مثال محلول بصيغة الاختبار

ما حل المعادلة −٣ك − ٥ = −٣٢؟
  1. أ) ك = ١١
  2. ب) ك = ٣٧٣
  3. ج) ك = ٩
  4. د) ك = −٩
  1. نضيف ٥ إلى الطرفين: −٣ك = −٢٧
  2. نقسم على −٣: ك = ٩
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما حل المعادلة ٧ت + ١١ = ٨١؟
الحل:
  1. نطرح ١١ من الطرفين: ٧ت = ٧٠
  2. نقسم على ٧: ت = ١٠
٢ما حل المعادلة −٣(س − ٦) = ٩؟
الحل:
  1. نقسم الطرفين على −٣: س − ٦ = −٣
  2. نضيف ٦: س = ٣
٣ما حل المعادلة −٤س + ٩ = ٣٣؟
الحل:
  1. نطرح ٩ من الطرفين: −٤س = ٢٤
  2. نقسم على −٤: س = −٦

متغير في طرفي المعادلة

١-٤

القاعدة

وزّع ⟸ اجمع حدود المتغير في طرف ⟸ حلّ. عبارة صحيحة دائمًا ⟸ متطابقة، وخاطئة ⟸ ∅.

مثال محلول بصيغة الاختبار

ما حل المعادلة ٤(م + ٤) = م + ١٩؟
  1. أ) م = ٣٥٣
  2. ب) م = ٣٥
  3. ج) م = ١
  4. د) م = −١
  1. خاصية التوزيع: ٤م + ١٦ = م + ١٩
  2. نطرح م من الطرفين: ٣م + ١٦ = ١٩
  3. نطرح ١٦: ٣م = ٣
  4. نقسم على ٣: م = ١
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما حل المعادلة ٤ن + ٤ = −٣ن + ١١؟
الحل:
  1. نضيف ٣ن إلى الطرفين: ٧ن + ٤ = ١١
  2. نطرح ٤: ٧ن = ٧
  3. نقسم على ٧: ن = ١
٢ما حل المعادلة −٤ك − ٥ = −٤ك − ٥؟
الحل:
  1. بعد التبسيط وطرح −٤ك من الطرفين يختفي المتغير.
  2. −٥ = −٥ عبارة صحيحة دائمًا ⟸ متطابقة.
٣ما حل المعادلة ٤(ن + ١) = ٥ن + ٨؟
الحل:
  1. خاصية التوزيع: ٤ن + ٤ = ٥ن + ٨
  2. نطرح ٥ن من الطرفين: −ن + ٤ = ٨
  3. نطرح ٤: −ن = ٤
  4. نقسم على −١: ن = −٤

معادلات القيمة المطلقة

١-٥

القاعدة

اعزل القيمة المطلقة ثم: |أ| = ب ⟸ أ = ب أو أ = −ب. إذا كانت ب سالبة ⟸ ∅.

مثال محلول بصيغة الاختبار

ما مجموعة حل المعادلة |٢ت − ٣| = ٧؟
  1. أ) {٥}
  2. ب) {−٢، ٥}
  3. ج) {−٥، ٢}
  4. د) {−٤، ١٠}
  1. ٢ت − ٣ = ٧ أو ٢ت − ٣ = −٧
  2. ٢ت = ١٠ أو ٢ت = −٤
  3. ت = ٥ أو ت = −٢
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما مجموعة حل المعادلة |ن − ٤| = ٥؟
الحل:
  1. ن − ٤ = ٥ أو ن − ٤ = −٥
  2. ن = ٩ أو ن = −١
٢ما مجموعة حل المعادلة |س + ٦| + ٥ = ٧؟
الحل:
  1. نعزل القيمة المطلقة: |س + ٦| = ٢
  2. س + ٦ = ٢ أو س + ٦ = −٢
  3. س = −٤ أو س = −٨
٣ما مجموعة حل المعادلة |م − ٤| = ٩؟
الحل:
  1. م − ٤ = ٩ أو م − ٤ = −٩
  2. م = ١٣ أو م = −٥

العلاقات: المجال والمدى والعكسية

٢-١

القاعدة

المجال: الإحداثيات الأولى، المدى: الثانية دون تكرار. العكسية: بدّل الإحداثيين.

مثال محلول بصيغة الاختبار

ما مدى العلاقة {(−١، −٤)، (−٥، ٥)، (٥، −٥)، (−١، ٨)}؟
  1. أ) {−٤، ٥}
  2. ب) {−٥، −٤، ٥، ٨}
  3. ج) {−٥، −١، ٥}
  4. د) {−٥، −٤، −١، ٥، ٨}
  1. المجال: مجموعة الإحداثيات الأولى (السينية)، والمدى: مجموعة الإحداثيات الثانية (الصادية)، ولا نكرر العنصر.
  2. المجال {−٥، −١، ٥} والمدى {−٥، −٤، ٥، ٨}.
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما العلاقة العكسية للعلاقة {(٤، ١)، (−٥، −٤)، (−٣، ٠)}؟
الحل:
  1. نحصل على العلاقة العكسية بتبديل الإحداثيين في كل زوج مرتب: (س، ص) ⟸ (ص، س).
  2. العكسية: {(١، ٤)، (−٤، −٥)، (٠، −٣)}
٢ما مدى العلاقة {(−١، −١)، (٥، ٠)، (−٣، −٢)، (−٢، ٢)}؟
الحل:
  1. المجال: مجموعة الإحداثيات الأولى (السينية)، والمدى: مجموعة الإحداثيات الثانية (الصادية)، ولا نكرر العنصر.
  2. المجال {−٣، −٢، −١، ٥} والمدى {−٢، −١، ٠، ٢}.
٣ما العلاقة العكسية للعلاقة {(٥، ٤)، (−٤، −٢)، (٦، ١)}؟
الحل:
  1. نحصل على العلاقة العكسية بتبديل الإحداثيين في كل زوج مرتب: (س، ص) ⟸ (ص، س).
  2. العكسية: {(٤، ٥)، (−٢، −٤)، (١، ٦)}

الدوال وقيمها

٢-٢

القاعدة

الدالة: لا يتكرر الإحداثي السيني بصورتين مختلفتين. د(أ): عوّض أ بدل س.

مثال محلول بصيغة الاختبار

إذا كانت د(س) = −٣س − ١، فما قيمة د(٤)؟
  1. أ) ٠
  2. ب) ١١
  3. ج) −١١
  4. د) −١٣
  1. نعوّض س = ٤: د(٤) = −٣(٤) − ١
  2. د(٤) = −١٣
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١أيّ العلاقات الآتية تمثل دالة؟
الحل:
  1. تكون العلاقة دالة إذا ارتبط كل عنصر في المجال بعنصر واحد فقط في المدى، أي لا يتكرر الإحداثي السيني مع إحداثيين صاديين مختلفين.
  2. في العلاقة {(−٥، −٣)، (١، −٤)، (٤، ٥)، (٠، −١)} لا يتكرر أي إحداثي سيني، أما بقية العلاقات ففيها إحداثي سيني مكرر.
٢إذا كانت د(س) = س² − ٥س − ٦، فما قيمة د(−٣)؟
الحل:
  1. نعوّض س = −٣: د(−٣) = (−٣)² − ٥(−٣) − ٦
  2. د(−٣) = ١٨
٣أيّ العلاقات الآتية تمثل دالة؟
الحل:
  1. تكون العلاقة دالة إذا ارتبط كل عنصر في المجال بعنصر واحد فقط في المدى، أي لا يتكرر الإحداثي السيني مع إحداثيين صاديين مختلفين.
  2. في العلاقة {(−٤، ٣)، (٤، ٥)، (٥، ٠)، (٣، −٤)} لا يتكرر أي إحداثي سيني، أما بقية العلاقات ففيها إحداثي سيني مكرر.

المقاطع وجذر الدالة

٢-٣ / ٢-٤

القاعدة

المقطع السيني: ص = ٠، الصادي: س = ٠. جذر الدالة = المقطع السيني = حل د(س) = ٠.

مثال محلول بصيغة الاختبار

ما المقطعان السيني والصادي للمستقيم س − ص = −٤؟
  1. أ) المقطع السيني ٤ ، المقطع الصادي −٤
  2. ب) المقطع السيني ١ ، المقطع الصادي −١
  3. ج) المقطع السيني −٤ ، المقطع الصادي −٤
  4. د) المقطع السيني −٤ ، المقطع الصادي ٤
  1. المقطع السيني: نضع ص = ٠: س = −٤
  2. المقطع الصادي: نضع س = ٠: −ص = −٤ ⟸ ص = ٤
  3. المستقيم يمر بالنقطتين (−٤، ٠) و (٠، ٤).
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما جذر الدالة د(س) = −٣س + ١٨ (حل المعادلة −٣س + ١٨ = ٠)؟
الحل:
  1. جذر الدالة هو قيمة س التي تجعل الدالة صفرًا، ويمثل المقطع السيني لتمثيلها البياني.
  2. −٣س + ١٨ = ٠ ⟸ −٣س = −١٨ ⟸ س = ٦
٢أيّ المعادلات الآتية خطية؟
الحل:
  1. المعادلة الخطية يمكن كتابتها بالصيغة القياسية أ س + ب ص = جـ: لا أسس للمتغيرات غير ١، ولا ضرب للمتغيرين، ولا متغير في المقام أو تحت الجذر.
  2. ص = س٣ + ٢ معادلة خطية، أما البقية ففيها س² أو س ص أو متغير في المقام أو تحت جذر.
٣ما المقطعان السيني والصادي للمستقيم س − ٥ص = −٥؟
الحل:
  1. المقطع السيني: نضع ص = ٠: س = −٥
  2. المقطع الصادي: نضع س = ٠: −٥ص = −٥ ⟸ ص = ١
  3. المستقيم يمر بالنقطتين (−٥، ٠) و (٠، ١).

الميل ومعدل التغير

٢-٥

القاعدة

م = ص٢ − ص١س٢ − س١، أفقي ⟸ ٠، رأسي ⟸ غير معرّف. معدل التغير = تغير الكمية ÷ تغير الزمن.

مثال محلول بصيغة الاختبار

ما ميل المستقيم المار بالنقطتين (٠، −١) و (−٢، −١)؟
  1. أ) ٠
  2. ب) −١
  3. ج) ١
  4. د) غير معرّف
  1. الميل م = ص٢ − ص١س٢ − س١
  2. م = −١ − (−١)−٢ − (٠) = ٠−٢ = ٠
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١كانت درجة الحرارة ١١° عند الساعة ٨ صباحًا، وأصبحت ١° بعد ٢ ساعات. ما معدل التغير في درجة الحرارة لكل ساعة؟
الحل:
  1. معدل التغير = التغير في الكميةالتغير في الزمن = ١ − ١١٢ = −١٠٢ = −٥ درجة لكل ساعة. (سالب ⟸ تناقص)
٢ما ميل المستقيم المار بالنقطتين (٠، −٦) و (−٢، −٦)؟
الحل:
  1. الميل م = ص٢ − ص١س٢ − س١
  2. م = −٦ − (−٦)−٢ − (٠) = ٠−٢ = ٠
٣كانت درجة الحرارة ١٧° عند الساعة ٨ صباحًا، وأصبحت ٢٩° بعد ٦ ساعات. ما معدل التغير في درجة الحرارة لكل ساعة؟
الحل:
  1. معدل التغير = التغير في الكميةالتغير في الزمن = ٢٩ − ١٧٦ = ١٢٦ = ٢ درجة لكل ساعة.

المتتابعات الحسابية

٢-٦

القاعدة

حن = ح١ + (ن − ١)د حيث د الفرق الثابت.

مثال محلول بصيغة الاختبار

ما الحد رقم ٢٣ في المتتابعة الحسابية ٢، −٣، −٨، −١٣، ...؟
  1. أ) −١٠٨
  2. ب) −١٠٣
  3. ج) −١١٥
  4. د) −١١٣
  1. أساس المتتابعة د = −٥ والحد الأول ح١ = ٢.
  2. حن = ح١ + (ن − ١)د ⟸ ح٢٣ = ٢ + (٢٣ − ١)(−٥) = −١٠٨
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما معادلة الحد النوني للمتتابعة الحسابية ٤، ٢، ٠، −٢، ...؟
الحل:
  1. حن = ح١ + (ن − ١)د حيث ح١ = ٤ و د = −٢.
  2. حن = ٤ + (ن − ١)(−٢) = −٢ن + ٦
٢ما الحد رقم ٢١ في المتتابعة الحسابية −١، −٤، −٧، −١٠، ...؟
الحل:
  1. أساس المتتابعة د = −٣ والحد الأول ح١ = −١.
