خصائص المساواة

  • ما تفعله بطرف افعله بالآخر.
  • معامل كسري ⟸ اضرب في المقلوب.

حالات خاصة

  • عبارة صحيحة دائمًا ⟸ متطابقة
  • عبارة خاطئة ⟸ ∅

القيمة المطلقة

  • |أ| = ب ⟸ أ = ±ب
  • |أ| = سالب ⟸ ∅

العلاقات والدوال

  • المجال: س ، المدى: ص
  • الدالة: لا تكرار للإحداثي السيني
  • د(أ): عوّض أ

المعادلة الخطية

  • أ س + ب ص = جـ
  • المقطع السيني ص = ٠، الصادي س = ٠
  • الجذر = المقطع السيني

الميل والمتتابعات

  • م = ص٢ − ص١س٢ − س١
  • حن = ح١ + (ن − ١)د

صيغ المستقيم

  • ص = م س + ب
  • ص − ص١ = م(س − س١)
  • أ س + ب ص = جـ

الميل

  • م = ص٢ − ص١س٢ − س١
  • من القياسية م = −أب
  • أفقي: ٠ — رأسي: غير معرّف

التوازي والتعامد

  • متوازيان: م١ = م٢
  • متعامدان: م١ × م٢ = −١

عكس رمز المتباينة

  • فقط عند الضرب/القسمة على سالب
  • ○ لـ < > و ● لـ ≤ ≥

المركبة

  • «و» ⟸ تقاطع ⟸ أ < س < ب
  • «أو» ⟸ اتحاد ⟸ شعاعان

القيمة المطلقة

  • |أ| < ب ⟸ −ب < أ < ب
  • |أ| > ب ⟸ أ < −ب أو أ > ب
  • |أ| < سالب ⟸ ∅

المعادلات ذات الخطوة الواحدة

١-٢

القاعدة

استعمل العملية العكسية للطرفين: الجمع ↔ الطرح، الضرب ↔ القسمة، ومع المعامل الكسري اضرب في المقلوب.

مثال محلول بصيغة الاختبار

ما حل المعادلة −٦س = −٤٢؟
  1. أ) س = ٥
  2. ب) س = ٨
  3. ج) س = −٧
  4. د) س = ٧
  1. نقسم الطرفين على −٦.
  2. س = ٧، وللتحقق نعوّض الحل في المعادلة الأصلية.
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما حل المعادلة ر + ١٠ = ٢١؟
الحل:
  1. نطرح ١٠ من الطرفين.
  2. ر = ١١، وللتحقق نعوّض الحل في المعادلة الأصلية.
٢ما حل المعادلة ن − ١١ = −١٠؟
الحل:
  1. نضيف ١١ إلى الطرفين.
  2. ن = ١، وللتحقق نعوّض الحل في المعادلة الأصلية.
٣ما حل المعادلة −٢٥ك = ٨؟
الحل:
  1. نضرب الطرفين في مقلوب المعامل −٥٢.
  2. ك = −٢٠، وللتحقق نعوّض الحل في المعادلة الأصلية.

المعادلات المتعددة الخطوات

١-٣

القاعدة

تخلّص من الحد الثابت أولًا ثم من المعامل. مع س + أب اضرب في ب أولًا.

مثال محلول بصيغة الاختبار

ما حل المعادلة ص٤ + ٧ = ١٢؟
  1. أ) ص = −٢٠
  2. ب) ص = ٢٠
  3. ج) ص = ٧٦
  4. د) ص = ٢٢
  1. نطرح ٧: ص٤ = ٥
  2. نضرب في ٤: ص = ٢٠
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما حل المعادلة ٢س − ١٠ = ٢؟
الحل:
  1. نضيف ١٠ إلى الطرفين: ٢س = ١٢
  2. نقسم على ٢: س = ٦
٢ما حل المعادلة −٣(ت + ٦) = −٦؟
الحل:
  1. نقسم الطرفين على −٣: ت + ٦ = ٢
  2. نطرح ٦: ت = −٤
٣ما حل المعادلة ٣(ر + ٤) = −٩؟
الحل:
  1. نقسم الطرفين على ٣: ر + ٤ = −٣
  2. نطرح ٤: ر = −٧

متغير في طرفي المعادلة

١-٤

القاعدة

وزّع ⟸ اجمع حدود المتغير في طرف ⟸ حلّ. عبارة صحيحة دائمًا ⟸ متطابقة، وخاطئة ⟸ ∅.

مثال محلول بصيغة الاختبار

ما حل المعادلة ٢م − ٦ = −٤م + ٤٢؟
  1. أ) م = −٨
  2. ب) م = ٨
  3. ج) م = ٦
  4. د) م = −٢٤
  1. نضيف ٤م إلى الطرفين: ٦م − ٦ = ٤٢
  2. نضيف ٦: ٦م = ٤٨
  3. نقسم على ٦: م = ٨
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما حل المعادلة ٣(ص − ٢) = −٢ص − ٦؟
الحل:
  1. خاصية التوزيع: ٣ص − ٦ = −٢ص − ٦
  2. نضيف ٢ص إلى الطرفين: ٥ص − ٦ = −٦
  3. نضيف ٦: ٥ص = ٠
  4. نقسم على ٥: ص = ٠
٢ما حل المعادلة ٢(ص + ١) = ٣ص − ٦؟
الحل:
  1. خاصية التوزيع: ٢ص + ٢ = ٣ص − ٦
  2. نطرح ٣ص من الطرفين: −ص + ٢ = −٦
  3. نطرح ٢: −ص = −٨
  4. نقسم على −١: ص = ٨
٣ما حل المعادلة −٥ر + ٥ = −٢ر + ٢؟
الحل:
  1. نضيف ٢ر إلى الطرفين: −٣ر + ٥ = ٢
  2. نطرح ٥: −٣ر = −٣
  3. نقسم على −٣: ر = ١

معادلات القيمة المطلقة

١-٥

القاعدة

اعزل القيمة المطلقة ثم: |أ| = ب ⟸ أ = ب أو أ = −ب. إذا كانت ب سالبة ⟸ ∅.

مثال محلول بصيغة الاختبار

ما مجموعة حل المعادلة |س − ١| + ٤ = ٨؟
  1. أ) {−٥، ٣}
  2. ب) {٥}
  3. ج) {−٤، ٤}
  4. د) {−٣، ٥}
  1. نعزل القيمة المطلقة: |س − ١| = ٤
  2. س − ١ = ٤ أو س − ١ = −٤
  3. س = ٥ أو س = −٣
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما مجموعة حل المعادلة |٢ك − ٣| = ٩؟
الحل:
  1. ٢ك − ٣ = ٩ أو ٢ك − ٣ = −٩
  2. ٢ك = ١٢ أو ٢ك = −٦
  3. ك = ٦ أو ك = −٣
٢ما مجموعة حل المعادلة |ص − ٦| = ٨؟
الحل:
  1. ص − ٦ = ٨ أو ص − ٦ = −٨
  2. ص = ١٤ أو ص = −٢
٣ما مجموعة حل المعادلة |٢س − ١| = ٥؟
الحل:
  1. ٢س − ١ = ٥ أو ٢س − ١ = −٥
  2. ٢س = ٦ أو ٢س = −٤
  3. س = ٣ أو س = −٢

الدوال وقيمها

٢-٢

القاعدة

الدالة: لا يتكرر الإحداثي السيني بصورتين مختلفتين. د(أ): عوّض أ بدل س.

مثال محلول بصيغة الاختبار

إذا كانت د(س) = ٤س + ٥، فما قيمة د(٣)؟
  1. أ) −٧
  2. ب) ٧
  3. ج) ١٢
  4. د) ١٧
  1. نعوّض س = ٣: د(٣) = ٤(٣) + ٥
  2. د(٣) = ١٧
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١أيّ العلاقات الآتية تمثل دالة؟
الحل:
  1. تكون العلاقة دالة إذا ارتبط كل عنصر في المجال بعنصر واحد فقط في المدى، أي لا يتكرر الإحداثي السيني مع إحداثيين صاديين مختلفين.
  2. في العلاقة {(٤، ٥)، (−١، ٣)، (٠، −٤)، (٦، ٧)} لا يتكرر أي إحداثي سيني، أما بقية العلاقات ففيها إحداثي سيني مكرر.
٢إذا كانت د(س) = ٢س − ٢، فما قيمة د(١)؟
الحل:
  1. نعوّض س = ١: د(١) = ٢(١) − ٢
  2. د(١) = ٠
٣أيّ العلاقات الآتية تمثل دالة؟
الحل:
  1. تكون العلاقة دالة إذا ارتبط كل عنصر في المجال بعنصر واحد فقط في المدى، أي لا يتكرر الإحداثي السيني مع إحداثيين صاديين مختلفين.
  2. في العلاقة {(−٣، ٥)، (٠، ٦)، (−٤، ٠)، (−١، ٦)} لا يتكرر أي إحداثي سيني، أما بقية العلاقات ففيها إحداثي سيني مكرر.

المقاطع وجذر الدالة

٢-٣ / ٢-٤

القاعدة

المقطع السيني: ص = ٠، الصادي: س = ٠. جذر الدالة = المقطع السيني = حل د(س) = ٠.

مثال محلول بصيغة الاختبار

ما المقطعان السيني والصادي للمستقيم ٢س − ص = −٤؟
  1. أ) المقطع السيني ٢ ، المقطع الصادي −٤
  2. ب) المقطع السيني ٤ ، المقطع الصادي −٢
  3. ج) المقطع السيني ٢ ، المقطع الصادي −١
  4. د) المقطع السيني −٢ ، المقطع الصادي ٤
  1. المقطع السيني: نضع ص = ٠: ٢س = −٤ ⟸ س = −٢
  2. المقطع الصادي: نضع س = ٠: −ص = −٤ ⟸ ص = ٤
  3. المستقيم يمر بالنقطتين (−٢، ٠) و (٠، ٤).
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما جذر الدالة د(س) = −٥س − ١٥ (حل المعادلة −٥س − ١٥ = ٠)؟
الحل:
  1. جذر الدالة هو قيمة س التي تجعل الدالة صفرًا، ويمثل المقطع السيني لتمثيلها البياني.
  2. −٥س − ١٥ = ٠ ⟸ −٥س = ١٥ ⟸ س = −٣
٢أيّ المعادلات الآتية خطية؟
الحل:
  1. المعادلة الخطية يمكن كتابتها بالصيغة القياسية أ س + ب ص = جـ: لا أسس للمتغيرات غير ١، ولا ضرب للمتغيرين، ولا متغير في المقام أو تحت الجذر.
  2. ٢س = ٥ معادلة خطية، أما البقية ففيها س² أو س ص أو متغير في المقام أو تحت جذر.
٣ما المقطعان السيني والصادي للمستقيم ٣س − ٢ص = ٦؟
الحل:
  1. المقطع السيني: نضع ص = ٠: ٣س = ٦ ⟸ س = ٢
  2. المقطع الصادي: نضع س = ٠: −٢ص = ٦ ⟸ ص = −٣
  3. المستقيم يمر بالنقطتين (٢، ٠) و (٠، −٣).

الميل ومعدل التغير

٢-٥

القاعدة

م = ص٢ − ص١س٢ − س١، أفقي ⟸ ٠، رأسي ⟸ غير معرّف. معدل التغير = تغير الكمية ÷ تغير الزمن.

مثال محلول بصيغة الاختبار

ما ميل المستقيم المار بالنقطتين (−١، ٠) و (−٤، −٣)؟
  1. أ) ٢
  2. ب) ٣٥
  3. ج) ١
  4. د) −١
  1. الميل م = ص٢ − ص١س٢ − س١
  2. م = −٣ − (٠)−٤ − (−١) = −٣−٣ = ١
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١كان في حصالة سلمى ٢٠٠ ريال، وبعد ٤ أسابيع أصبح فيها ٢٨٠ ريالًا. ما معدل التغير لكل أسبوع؟
الحل:
  1. معدل التغير = التغير في الكميةالتغير في الزمن = ٢٨٠ − ٢٠٠٤ = ٨٠٤ = ٢٠ ريال لكل أسبوع.
٢ما ميل المستقيم المار بالنقطتين (−١، ٥) و (−٤، ١٣)؟
الحل:
  1. الميل م = ص٢ − ص١س٢ − س١
  2. م = ١٣ − (٥)−٤ − (−١) = ٨−٣ = −٨٣
٣ارتفعت طائرة من ٥٠٠ م إلى ٢٩٠٠ م خلال ٦ دقائق. ما معدل التغير في الارتفاع لكل دقيقة؟
الحل:
  1. معدل التغير = التغير في الكميةالتغير في الزمن = ٢٩٠٠ − ٥٠٠٦ = ٢٤٠٠٦ = ٤٠٠ م لكل دقيقة.

المتتابعات الحسابية

٢-٦

القاعدة

حن = ح١ + (ن − ١)د حيث د الفرق الثابت.

مثال محلول بصيغة الاختبار

ما الحد رقم ٢٣ في المتتابعة الحسابية −٥، ١، ٧، ١٣، ...؟
  1. أ) ١٣٣
  2. ب) ١٣٨
  3. ج) ١٢١
  4. د) ١٢٧
  1. أساس المتتابعة د = ٦ والحد الأول ح١ = −٥.
  2. حن = ح١ + (ن − ١)د ⟸ ح٢٣ = −٥ + (٢٣ − ١)(٦) = ١٢٧
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما معادلة الحد النوني للمتتابعة الحسابية ٤، ٨، ١٢، ١٦، ...؟
الحل:
  1. حن = ح١ + (ن − ١)د حيث ح١ = ٤ و د = ٤.
  2. حن = ٤ + (ن − ١)(٤) = ٤ن
٢ما الحد رقم ١٧ في المتتابعة الحسابية ١٣، ١٩، ٢٥، ٣١، ...؟
الحل:
  1. أساس المتتابعة د = ٦ والحد الأول ح١ = ١٣.
  2. حن = ح١ + (ن − ١)د ⟸ ح١٧ = ١٣ + (١٧ − ١)(٦) = ١٠٩
٣ما معادلة الحد النوني للمتتابعة الحسابية −١، ٠، ١، ٢، ...؟
الحل:
  1. حن = ح١ + (ن − ١)د حيث ح١ = −١ و د = ١.
  2. حن = −١ + (ن − ١)(١) = ن − ٢

الميل والمقطع الصادي من المعادلة

٣-١

القاعدة

اعزل ص لتصبح المعادلة ص = م س + ب؛ معامل س هو الميل والحد الثابت هو المقطع. من أ س + ب ص = جـ: م = −أب و ب = جـب.

مثال محلول بصيغة الاختبار

ما الميل والمقطع الصادي للمستقيم ٦س + ص = −٤؟
  1. أ) الميل = ٦ ، المقطع الصادي = ٤
  2. ب) الميل = −٦ ، المقطع الصادي = ٤
  3. ج) الميل = ٦ ، المقطع الصادي = −٤
  4. د) الميل = −٦ ، المقطع الصادي = −٤
  1. نكتب المعادلة بصيغة الميل والمقطع ص = م س + ب بعزل ص.
  2. نطرح ٦س من الطرفين: ص = −٦س − ٤
  3. إذن الميل م = −٦ والمقطع الصادي ب = −٤.
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما الميل والمقطع الصادي للمستقيم ٣س − ص = ١؟
الحل:
  1. نكتب المعادلة بصيغة الميل والمقطع ص = م س + ب بعزل ص.
  2. نطرح ٣س من الطرفين: −ص = −٣س + ١
  3. نقسم الطرفين على −١: ص = ٣س − ١
  4. إذن الميل م = ٣ والمقطع الصادي ب = −١.
٢ما الميل والمقطع الصادي للمستقيم ٥س + ٢ص = ٤؟
الحل:
  1. نكتب المعادلة بصيغة الميل والمقطع ص = م س + ب بعزل ص.
  2. نطرح ٥س من الطرفين: ٢ص = −٥س + ٤
  3. نقسم الطرفين على ٢: ص = −٥٢س + ٢
  4. إذن الميل م = −٥٢ والمقطع الصادي ب = ٢.
٣ما الميل والمقطع الصادي للمستقيم ٤س + ص = ٣؟
الحل:
  1. نكتب المعادلة بصيغة الميل والمقطع ص = م س + ب بعزل ص.
  2. نطرح ٤س من الطرفين: ص = −٤س + ٣
  3. إذن الميل م = −٤ والمقطع الصادي ب = ٣.

معادلة مستقيم بمعلومية ميله ونقطة

٣-٢

القاعدة

ما معادلة المستقيم الذي ميله ٢٣ ويمر بالنقطة (٣، −٤) بصيغة الميل والمقطع؟

  2. −٤ = ٢ + ب ⟸ ب = −٦

صيغة الميل ونقطة والتحويل بين الصيغ

٣-٣

القاعدة

ص − ص١ = م(س − س١): إشارة الإحداثي داخل الصيغة معكوسة. للتحويل: وزّع ثم اعزل ص، وللصيغة القياسية تخلّص من الكسور واجعل أ ≥ ٠.

مثال محلول بصيغة الاختبار

ما معادلة المستقيم المار بالنقطة (−٢، −١) وميله −١ بصيغة الميل ونقطة؟
  1. أ) ص + ١ = س + ٢
  2. ب) ص + ٢ = −(س + ١)
  3. ج) ص − ١ = −(س − ٢)
  4. د) ص + ١ = −(س + ٢)
  1. صيغة الميل ونقطة: ص − ص١ = م(س − س١)
  2. نعوّض س١ = −٢ ، ص١ = −١ ، م = −١
  3. ص − (−١) = −١(س − (−٢)) ⟸ ص + ١ = −(س + ٢)
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١اكتب المعادلة ص − ١ = ١٢(س + ٦) بصيغة الميل والمقطع.
الحل:
  1. نوزّع: ص − ١ = ١٢س + ٣
  2. نضيف ١ إلى الطرفين: ص = ١٢س + ٤

الحل:
  1. نضرب الطرفين في ٥ للتخلص من الكسر: ٥ص = ٣س + ١٥
  2. ننقل حد س إلى الطرف الأيسر ونجعل أ موجبًا: ٣س − ٥ص = −١٥
  3. تحقق: الصيغة القياسية تتطلب أ ≥ ٠ وأعدادًا صحيحة بلا كسور.
٣ما معادلة المستقيم المار بالنقطة (−٣، ٦) وميله ٣ بصيغة الميل ونقطة؟
الحل:
  1. صيغة الميل ونقطة: ص − ص١ = م(س − س١)
  2. نعوّض س١ = −٣ ، ص١ = ٦ ، م = ٣
  3. ص − (٦) = ٣(س − (−٣)) ⟸ ص − ٦ = ٣(س + ٣)

  2. نعوّض النقطة: ٤ = −١٣(−٣) + ب ⟸ ب = ٣

الحل:
  1. ميل الأول −١.
  2. نعيد كتابة الثاني: ص = س − ٤ ⟸ ميله ١.
  3. حاصل ضرب الميلين −١ × ١ = −١ ⟸ متعامدان.
٣ما معادلة المستقيم المار بالنقطة (٤، ٢) والموازي للمستقيم ص = −١٤س + ٤؟
الحل:

  2. نعوّض النقطة: ٢ = −١٤(٤) + ب ⟸ ب = ٣
  3. المعادلة: ص = −١٤س + ٣

القاعدة

  4. د) ك ≥ ٥

المتباينات المركبة

٤-٤

القاعدة

«و»: نفّذ العملية على الأجزاء الثلاثة، والحل تقاطع (قطعة). «أو»: حل كل متباينة منفردة، والحل اتحاد (شعاعان).

ما حل المتباينة المركبة −٨ ≤ س − ٥ < −٦؟

  2. ب) −٣ ≤ س < −١
  3. ج) س < −٣ أو س > −١
  4. د) −١٣ ≤ س < −١١
  1. المتباينة المركبة المتضمنة «و» تُحل بإجراء العملية نفسها على الأجزاء الثلاثة.
  2. نضيف ٥ إلى الأجزاء الثلاثة: −٣ ≤ س < −١
  3. الحل هو تقاطع الشرطين: جميع الأعداد بين −٣ و −١.
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما حل المتباينة المركبة س + ٨ < ٩ أو ٣س − ٢ ≥ ١٠؟
الحل:
  1. نحل كل متباينة على حدة، ثم نأخذ الاتحاد (أي عدد يحقق إحداهما على الأقل).

  3. الثانية: نضيف ٢ ثم نقسم على ٣ ⟸ س ≥ ٤
  4. الحل: س < ١ أو س ≥ ٤
٢ما حل المتباينة المركبة −٨ < ٣ن + ١ ≤ ٤؟
الحل:
  1. المتباينة المركبة المتضمنة «و» تُحل بإجراء العملية نفسها على الأجزاء الثلاثة.
  2. نطرح ١ من الأجزاء الثلاثة: −٩ < ٣ن ≤ ٣
  3. نقسم الأجزاء الثلاثة على ٣: −٣ < ن ≤ ١
  4. الحل هو تقاطع الشرطين: جميع الأعداد بين −٣ و ١.
٣ما حل المتباينة المركبة ك + ٦ ≤ ٥ أو ٢ك − ٦ > −٢؟
