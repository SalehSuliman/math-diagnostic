📐 ورقة القوانين السريعة

هذه المراجعة مبنية على جدول مواصفات الاختبار المقابل: كل قسم يقابل رقم سؤال في نماذج الاختبار، ويحتوي القاعدة ومثالًا محلولًا بالصيغة نفسها وتدريبات تفاعلية.

صيغ المستقيم

  • ص = م س + ب
  • ص − ص١ = م(س − س١)
  • أ س + ب ص = جـ

الميل

  • م = ص٢ − ص١س٢ − س١
  • من القياسية م = −أب
  • أفقي: ٠ — رأسي: غير معرّف

التوازي والتعامد

  • متوازيان: م١ = م٢
  • متعامدان: م١ × م٢ = −١

عكس رمز المتباينة

  • فقط عند الضرب/القسمة على سالب
  • ○ لـ < > و ● لـ ≤ ≥

المركبة

  • «و» ⟸ تقاطع ⟸ أ < س < ب
  • «أو» ⟸ اتحاد ⟸ شعاعان

القيمة المطلقة

  • |أ| < ب ⟸ −ب < أ < ب
  • |أ| > ب ⟸ أ < −ب أو أ > ب
  • |أ| < سالب ⟸ ∅
س١

الميل والمقطع الصادي من المعادلة

٣-١

القاعدة

اعزل ص لتصبح المعادلة ص = م س + ب؛ معامل س هو الميل والحد الثابت هو المقطع. من أ س + ب ص = جـ: م = −أب و ب = جـب.

مثال محلول بصيغة الاختبار

ما الميل والمقطع الصادي للمستقيم ٣س + ٤ص = ٤؟
  1. أ) الميل = −٣٤ ، المقطع الصادي = ١
  2. ب) الميل = ٣٤ ، المقطع الصادي = ١
  3. ج) الميل = ٣ ، المقطع الصادي = ٤
  4. د) الميل = −٣٤ ، المقطع الصادي = −١
  1. نكتب المعادلة بصيغة الميل والمقطع ص = م س + ب بعزل ص.
  2. نطرح ٣س من الطرفين: ٤ص = −٣س + ٤
  3. نقسم الطرفين على ٤: ص = −٣٤س + ١
  4. إذن الميل م = −٣٤ والمقطع الصادي ب = ١.
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما الميل والمقطع الصادي للمستقيم س + ٤ص = −١٢؟
الحل:
  1. نكتب المعادلة بصيغة الميل والمقطع ص = م س + ب بعزل ص.
  2. نطرح س من الطرفين: ٤ص = −س − ١٢
  3. نقسم الطرفين على ٤: ص = −١٤س − ٣
  4. إذن الميل م = −١٤ والمقطع الصادي ب = −٣.
٢ما الميل والمقطع الصادي للمستقيم ٢س + ٣ص = −٣؟
الحل:
  1. نكتب المعادلة بصيغة الميل والمقطع ص = م س + ب بعزل ص.
  2. نطرح ٢س من الطرفين: ٣ص = −٢س − ٣
  3. نقسم الطرفين على ٣: ص = −٢٣س − ١
  4. إذن الميل م = −٢٣ والمقطع الصادي ب = −١.
٣ما الميل والمقطع الصادي للمستقيم س + ٤ص = −٢٠؟
الحل:
  1. نكتب المعادلة بصيغة الميل والمقطع ص = م س + ب بعزل ص.
  2. نطرح س من الطرفين: ٤ص = −س − ٢٠
  3. نقسم الطرفين على ٤: ص = −١٤س − ٥
  4. إذن الميل م = −١٤ والمقطع الصادي ب = −٥.

نمذجة موقف حياتي بمعادلة خطية

٣-١

القاعدة

القيمة الثابتة/الابتدائية ⟸ المقطع ب، والمعدل لكل وحدة ⟸ الميل م (سالب عند التناقص).

مثال محلول بصيغة الاختبار

خزان فيه ٤٠٠ لتر من الماء، ويُستهلك منه ٢٥ لترًا يوميًّا. ما المعادلة التي تمثل كمية الماء ص بعد س يومًا؟
  1. أ) ص = ٤٠٠س − ٢٥
  2. ب) ص = −٢٥س − ٤٠٠
  3. ج) ص = −٢٥س + ٤٠٠
  4. د) ص = ٢٥س + ٤٠٠
  1. نحدد القيمة الثابتة (لا تتغير مع س) = المقطع الصادي ب = ٤٠٠، والمعدّل (يتغير مع كل وحدة) = الميل م = −٢٥.
  2. المعدل سالب لأن الكمية تتناقص. المعادلة: ص = −٢٥س + ٤٠٠
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما معادلة المستقيم الذي ميله ٢ ومقطعه الصادي ٦؟
الحل:
  1. صيغة الميل والمقطع: ص = م س + ب حيث م الميل وب المقطع الصادي.
  2. نعوّض م = ٢ و ب = ٦: ص = ٢س + ٦
٢رسوم التسجيل في نادٍ رياضي ٢٠٠ ريال، والاشتراك الشهري ١٢٠ ريالًا. ما المعادلة التي تمثل التكلفة الكلية ص بعد س شهرًا؟
الحل:
  1. نحدد القيمة الثابتة (لا تتغير مع س) = المقطع الصادي ب = ٢٠٠، والمعدّل (يتغير مع كل وحدة) = الميل م = ١٢٠.
  2. المعادلة: ص = ١٢٠س + ٢٠٠
٣ما معادلة المستقيم الذي ميله ١ ومقطعه الصادي ٤؟
الحل:
  1. صيغة الميل والمقطع: ص = م س + ب حيث م الميل وب المقطع الصادي.
  2. نعوّض م = ١ و ب = ٤: ص = س + ٤
س٣

معادلة مستقيم بمعلومية ميله ونقطة

٣-٢

القاعدة

عوّض الميل والنقطة في ص = م س + ب وأوجد ب. انتبه لضرب الأعداد السالبة.

مثال محلول بصيغة الاختبار

ما معادلة المستقيم الذي ميله ٣ ويمر بالنقطة (٣، ٥) بصيغة الميل والمقطع؟
  1. أ) ص = −٣س − ٤
  2. ب) ص = ٣س + ٥
  3. ج) ص = ٣س + ١٤
  4. د) ص = ٣س − ٤
  1. نعوّض الميل والنقطة في ص = م س + ب: ٥ = ٣(٣) + ب
  2. ٥ = ٩ + ب ⟸ ب = −٤
  3. المعادلة: ص = ٣س − ٤
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما معادلة المستقيم الذي ميله −٢ ويمر بالنقطة (−٦، ١) بصيغة الميل والمقطع؟
الحل:
  1. نعوّض الميل والنقطة في ص = م س + ب: ١ = −٢(−٦) + ب
  2. ١ = ١٢ + ب ⟸ ب = −١١
  3. المعادلة: ص = −٢س − ١١
٢ما معادلة المستقيم الذي ميله ١٢ ويمر بالنقطة (−٦، −١) بصيغة الميل والمقطع؟
الحل:
  1. نعوّض الميل والنقطة في ص = م س + ب: −١ = ١٢(−٦) + ب
  2. −١ = −٣ + ب ⟸ ب = ٢
  3. المعادلة: ص = ١٢س + ٢
٣ما معادلة المستقيم الذي ميله −٤ ويمر بالنقطة (−٣، −٤) بصيغة الميل والمقطع؟
الحل:
  1. نعوّض الميل والنقطة في ص = م س + ب: −٤ = −٤(−٣) + ب
  2. −٤ = ١٢ + ب ⟸ ب = −١٦
  3. المعادلة: ص = −٤س − ١٦
س٤

معادلة المستقيم المار بنقطتين

٣-٢

القاعدة

احسب م = ص٢ − ص١س٢ − س١ ثم عوّض بإحدى النقطتين لإيجاد ب.

مثال محلول بصيغة الاختبار

ما معادلة المستقيم المار بالنقطتين (٢، ١) و (٣، ٣)؟
  1. أ) ص = ١٢س
  2. ب) ص = ٢س + ٥
  3. ج) ص = −٢س + ٥
  4. د) ص = ٢س − ٣
  1. الميل: م = ص٢ − ص١س٢ − س١ = ٣ − (١)٣ − (٢) = ٢١ = ٢
  2. نعوّض النقطة (٢، ١): ١ = ٢(٢) + ب ⟸ ب = −٣
  3. المعادلة: ص = ٢س − ٣
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما معادلة المستقيم المار بالنقطتين (٣، −٢) و (٦، −٨)؟
الحل:
  1. الميل: م = ص٢ − ص١س٢ − س١ = −٨ − (−٢)٦ − (٣) = −٦٣ = −٢
  2. نعوّض النقطة (٣، −٢): −٢ = −٢(٣) + ب ⟸ ب = ٤
  3. المعادلة: ص = −٢س + ٤
٢ما معادلة المستقيم المار بالنقطتين (٠، ١) و (٢، ٧)؟
الحل:
  1. الميل: م = ص٢ − ص١س٢ − س١ = ٧ − (١)٢ − (٠) = ٦٢ = ٣
  2. نعوّض النقطة (٠، ١): ١ = ٣(٠) + ب ⟸ ب = ١
  3. المعادلة: ص = ٣س + ١
٣ما معادلة المستقيم المار بالنقطتين (−١، ٠) و (٢، ٢)؟
الحل:
  1. الميل: م = ص٢ − ص١س٢ − س١ = ٢ − (٠)٢ − (−١) = ٢٣ = ٢٣
  2. نعوّض النقطة (−١، ٠): ٠ = ٢٣(−١) + ب ⟸ ب = ٢٣
  3. المعادلة: ص = ٢٣س + ٢٣
س٥

صيغة الميل ونقطة والتحويل بين الصيغ

٣-٣

القاعدة

ص − ص١ = م(س − س١): إشارة الإحداثي داخل الصيغة معكوسة. للتحويل: وزّع ثم اعزل ص، وللصيغة القياسية تخلّص من الكسور واجعل أ ≥ ٠.

مثال محلول بصيغة الاختبار

ما معادلة المستقيم المار بالنقطة (−٦، −٦) وميله ٤ بصيغة الميل ونقطة؟
  1. أ) ص + ٦ = ٤(س + ٦)
  2. ب) ص − ٦ = ٤(س − ٦)
  3. ج) ص + ٦ = −٤(س + ٦)
  4. د) ص + ٦ = ٤(س − ٦)
  1. صيغة الميل ونقطة: ص − ص١ = م(س − س١)
  2. نعوّض س١ = −٦ ، ص١ = −٦ ، م = ٤
  3. ص − (−٦) = ٤(س − (−٦)) ⟸ ص + ٦ = ٤(س + ٦)
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١اكتب المعادلة ص − ٤ = س + ٦ بصيغة الميل والمقطع.
الحل:
  1. نوزّع: ص − ٤ = س + ٦
  2. نضيف ٤ إلى الطرفين: ص = س + ١٠
٢اكتب المعادلة ص = −٢٥س + ٤ بالصيغة القياسية أ س + ب ص = جـ.
الحل:
  1. نضرب الطرفين في ٥ للتخلص من الكسر: ٥ص = −٢س + ٢٠
  2. ننقل حد س إلى الطرف الأيسر ونجعل أ موجبًا: ٢س + ٥ص = ٢٠
  3. تحقق: الصيغة القياسية تتطلب أ ≥ ٠ وأعدادًا صحيحة بلا كسور.
٣ما معادلة المستقيم المار بالنقطة (٦، ٣) وميله −٢ بصيغة الميل ونقطة؟
الحل:
  1. صيغة الميل ونقطة: ص − ص١ = م(س − س١)
  2. نعوّض س١ = ٦ ، ص١ = ٣ ، م = −٢
  3. ص − (٣) = −٢(س − (٦)) ⟸ ص − ٣ = −٢(س − ٦)

المستقيمات المتوازية والمتعامدة

٣-٤

القاعدة

التوازي: الميل نفسه. التعامد: المعكوس السالب (م١ × م٢ = −١). ثم عوّض النقطة لإيجاد ب.

مثال محلول بصيغة الاختبار

ما معادلة المستقيم المار بالنقطة (٣، −٥) والموازي للمستقيم ص = ١٣س − ٣؟
  1. أ) ص = ١٣س − ٦
  2. ب) ص = −٣س + ٤
  3. ج) ص = −١٣س − ٤
  4. د) ص = ١٣س − ٣
  1. المستقيمان المتوازيان لهما الميل نفسه، إذن م = ١٣.
  2. نعوّض النقطة: −٥ = ١٣(٣) + ب ⟸ ب = −٦
  3. المعادلة: ص = ١٣س − ٦
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما معادلة المستقيم المار بالنقطة (−٦، ٠) والعمودي على المستقيم ص = ٢س − ٥؟
الحل:
  1. ميل المستقيم المعطى ٢، وميل العمودي عليه هو المعكوس السالب: م = −١٢ (لأن ٢ × −١٢ = −١).
  2. نعوّض النقطة: ٠ = −١٢(−٦) + ب ⟸ ب = −٣
  3. المعادلة: ص = −١٢س − ٣
٢ما العلاقة بين المستقيمين ص = ٣٤س + ٣ و ٣س − ٤ص = −٢٤؟
الحل:
  1. ميل الأول ٣٤.
  2. نعيد كتابة الثاني: ص = ٣٤س + ٦ ⟸ ميله ٣٤.
  3. الميلان متساويان والمقطعان مختلفان ⟸ المستقيمان متوازيان.
٣ما معادلة المستقيم المار بالنقطة (−٣، −٥) والموازي للمستقيم ص = −٢س + ٣؟
الحل:
  1. المستقيمان المتوازيان لهما الميل نفسه، إذن م = −٢.
  2. نعوّض النقطة: −٥ = −٢(−٣) + ب ⟸ ب = −١١
  3. المعادلة: ص = −٢س − ١١

حل متباينة بخطوة واحدة

٤-١ / ٤-٢

القاعدة

الجمع والطرح والضرب/القسمة على موجب لا تغيّر الرمز. الضرب أو القسمة على سالب تعكس الرمز.

مثال محلول بصيغة الاختبار

ما حل المتباينة ت + ٥ < −٩؟
  1. أ) ت < ١٤
  2. ب) ت > −١٤
  3. ج) ت < −١٤
  4. د) ت < −٤
  1. نطرح ٥ من الطرفين (لا يتغير اتجاه رمز المتباينة عند الجمع أو الطرح).
  2. ت < −١٤ ، ومجموعة الحل {ت | ت < −١٤}
  3. التمثيل: دائرة مفتوحة عند −١٤ وسهم نحو اليسار.
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما حل المتباينة −س٥ ≥ ١؟
الحل:
  1. نضرب الطرفين في −٥ ونعكس رمز المتباينة.
  2. س ≤ −٥ ، ومجموعة الحل {س | س ≤ −٥}
٢ما حل المتباينة ت − ٨ ≤ ٧؟
الحل:
  1. نضيف ٨ إلى الطرفين (لا يتغير اتجاه رمز المتباينة عند الجمع أو الطرح).
  2. ت ≤ ١٥ ، ومجموعة الحل {ت | ت ≤ ١٥}
  3. التمثيل: دائرة مغلقة عند ١٥ وسهم نحو اليسار.
٣ما حل المتباينة −٨ر < ٣٢؟
الحل:
  1. نقسم الطرفين على −٨ ونعكس رمز المتباينة لأننا قسمنا على عدد سالب.
  2. ر > −٤ ، ومجموعة الحل {ر | ر > −٤}

المتباينات المتعددة الخطوات

٤-٣

القاعدة

وزّع ⟸ اجمع حدود المتغير في طرف ⟸ اعزل ⟸ اقسم واعكس عند السالب. في المسائل اللفظية خذ أكبر/أصغر عدد صحيح مناسب.

مثال محلول بصيغة الاختبار

ما حل المتباينة ٤ت − ٨ > −٣ت − ٣٦؟
  1. أ) ت > −٤
  2. ب) ت < −٤
  3. ج) ت > −٤٤٧
  4. د) ت > ٤
  1. نضيف ٣ت إلى الطرفين: ٧ت − ٨ > −٣٦
  2. نضيف ٨ إلى الطرفين: ٧ت > −٢٨
  3. نقسم على ٧: ت > −٤
  4. مجموعة الحل: {ت | ت > −٤}
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١مع خالد ٣٥٠ ريال، ويريد شراء كتابًا سعر الكتاب ٥٥ ريالًا مع رسوم توصيل ثابتة ١٥ ريالًا. ما أكبر عدد يمكنه شراؤه؟
الحل:
  1. نفرض العدد ن، والمتباينة: ٥٥ن + ١٥ ≤ ٣٥٠
  2. ٥٥ن ≤ ٣٣٥ ⟸ ن ≤ ٣٣٥٥٥ ≈ ٦٫٠٩
  3. العدد يجب أن يكون صحيحًا، فأكبر قيمة هي ٦.
٢ما حل المتباينة ٤(ص + ٤) > ٢ص + ٨؟
الحل:
  1. نستعمل خاصية التوزيع: ٤ص + ١٦ > ٢ص + ٨
  2. نطرح ٢ص من الطرفين: ٢ص + ١٦ > ٨
  3. نطرح ١٦ من الطرفين: ٢ص > −٨
  4. نقسم على ٢: ص > −٤
  5. مجموعة الحل: {ص | ص > −٤}
٣مع خالد ٢٥٠ ريال، ويريد شراء قميصًا سعر القميص ٤٠ ريالًا مع رسوم توصيل ثابتة ١٥ ريالًا. ما أكبر عدد يمكنه شراؤه؟
الحل:
  1. نفرض العدد ن، والمتباينة: ٤٠ن + ١٥ ≤ ٢٥٠
  2. ٤٠ن ≤ ٢٣٥ ⟸ ن ≤ ٢٣٥٤٠ ≈ ٥٫٨٨
  3. العدد يجب أن يكون صحيحًا، فأكبر قيمة هي ٥.
س٩

المتباينات المركبة

٤-٤

القاعدة

«و»: نفّذ العملية على الأجزاء الثلاثة، والحل تقاطع (قطعة). «أو»: حل كل متباينة منفردة، والحل اتحاد (شعاعان).

مثال محلول بصيغة الاختبار

ما حل المتباينة المركبة −٢ < ن − ٣ < ١؟
  1. أ) −٥ < ن < −٢
  2. ب) ١ < ن < ٤
  3. ج) ن < ١ أو ن > ٤
  4. د) ١ ≤ ن ≤ ٤
  1. المتباينة المركبة المتضمنة «و» تُحل بإجراء العملية نفسها على الأجزاء الثلاثة.
  2. نضيف ٣ إلى الأجزاء الثلاثة: ١ < ن < ٤
  3. الحل هو تقاطع الشرطين: جميع الأعداد بين ١ و ٤.
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما حل المتباينة المركبة ت + ٥ < ٣ أو ٣ت − ٣ > ١٢؟
الحل:
  1. نحل كل متباينة على حدة، ثم نأخذ الاتحاد (أي عدد يحقق إحداهما على الأقل).
  2. الأولى: نطرح ٥ ⟸ ت < −٢
  3. الثانية: نضيف ٣ ثم نقسم على ٣ ⟸ ت > ٥
  4. الحل: ت < −٢ أو ت > ٥
٢ما حل المتباينة المركبة ٨ ≤ ٣ص + ٥ ≤ ٢٠؟
الحل:
  1. المتباينة المركبة المتضمنة «و» تُحل بإجراء العملية نفسها على الأجزاء الثلاثة.
  2. نطرح ٥ من الأجزاء الثلاثة: ٣ ≤ ٣ص ≤ ١٥
  3. نقسم الأجزاء الثلاثة على ٣: ١ ≤ ص ≤ ٥
  4. الحل هو تقاطع الشرطين: جميع الأعداد بين ١ و ٥.
٣ما حل المتباينة المركبة ك + ٤ ≤ ١ أو ٢ك − ٥ > ١؟
الحل:
  1. نحل كل متباينة على حدة، ثم نأخذ الاتحاد (أي عدد يحقق إحداهما على الأقل).
  2. الأولى: نطرح ٤ ⟸ ك ≤ −٣
  3. الثانية: نضيف ٥ ثم نقسم على ٢ ⟸ ك > ٣
  4. الحل: ك ≤ −٣ أو ك > ٣
س١٠

متباينات القيمة المطلقة

٤-٥

القاعدة

|أ| < ب ⟸ −ب < أ < ب (و). |أ| > ب ⟸ أ < −ب أو أ > ب (أو). القيمة المطلقة لا تكون سالبة.

مثال محلول بصيغة الاختبار

ما حل المتباينة |س + ٣| < ٦؟
  1. أ) س < −٩ أو س > ٣
  2. ب) −٩ < س < ٣
  3. ج) −٦ < س < ٦
  4. د) −٣ < س < ٩
  1. |أ| < ب (حيث ب > ٠) تكافئ متباينة «و»: −٦ < أ < ٦
  2. −٦ < س + ٣ < ٦
  3. نطرح ٣ للأجزاء الثلاثة: −٩ < س < ٣
  4. المعنى: المسافة بين س و −٣ أقل من ٦.
تدرّب (أسئلة مشابهة لأسئلة الاختبار)
١ما حل المتباينة |ت + ٥| ≥ ٥؟
الحل:
  1. |أ| ≥ ب (حيث ب > ٠) تكافئ متباينة «أو»: أ ≤ −٥ أو أ ≥ ٥
  2. ت + ٥ ≤ −٥ أو ت + ٥ ≥ ٥
  3. نحل كلًّا منهما: ت ≤ −١٠ أو ت ≥ ٠
٢ما حل المتباينة |ر − ٣| ≤ ٥؟
الحل:
  1. |أ| ≤ ب (حيث ب > ٠) تكافئ متباينة «و»: −٥ ≤ أ ≤ ٥
  2. −٥ ≤ ر − ٣ ≤ ٥
  3. نضيف ٣ للأجزاء الثلاثة: −٢ ≤ ر ≤ ٨
  4. المعنى: المسافة بين ر و ٣ أقل من أو تساوي ٥.
٣ما حل المتباينة |س| > ٣؟
الحل:
  1. |أ| > ب (حيث ب > ٠) تكافئ متباينة «أو»: أ < −٣ أو أ > ٣
  2. س < −٣ أو س > ٣
  3. نحل كلًّا منهما: س < −٣ أو س > ٣

🧪 اختبار تجريبي ١ (محاكٍ للاختبار)

١٠ سؤالًا
١ما الميل والمقطع الصادي للمستقيم ٣س − ٤ص = −١٢؟
الحل:
  1. نكتب المعادلة بصيغة الميل والمقطع ص = م س + ب بعزل ص.
  2. نطرح ٣س من الطرفين: −٤ص = −٣س − ١٢
  3. نقسم الطرفين على −٤: ص = ٣٤س + ٣
  4. إذن الميل م = ٣٤ والمقطع الصادي ب = ٣.
٢تتقاضى شركة سيارات أجرة ٥ ريالات أجرة ثابتة و٢ ريالات عن كل كيلومتر. ما المعادلة التي تمثل التكلفة ص لرحلة طولها س كيلومترًا؟
الحل:
  1. نحدد القيمة الثابتة (لا تتغير مع س) = المقطع الصادي ب = ٥، والمعدّل (يتغير مع كل وحدة) = الميل م = ٢.
  2. المعادلة: ص = ٢س + ٥
٣ما معادلة المستقيم الذي ميله −١ ويمر بالنقطة (١، −٦) بصيغة الميل والمقطع؟
الحل:
  1. نعوّض الميل والنقطة في ص = م س + ب: −٦ = −١(١) + ب
  2. −٦ = −١ + ب ⟸ ب = −٥
  3. المعادلة: ص = −س − ٥
٤ما معادلة المستقيم المار بالنقطتين (−٥، ٣) و (−١، ٧)؟
الحل:
  1. الميل: م = ص٢ − ص١س٢ − س١ = ٧ − (٣)−١ − (−٥) = ٤٤ = ١
  2. نعوّض النقطة (−٥، ٣): ٣ = ١(−٥) + ب ⟸ ب = ٨
  3. المعادلة: ص = س + ٨
٥اكتب المعادلة ص − ١ = ٤(س + ٢) بصيغة الميل والمقطع.
الحل:
  1. نوزّع: ص − ١ = ٤س + ٨
  2. نضيف ١ إلى الطرفين: ص = ٤س + ٩
٦ما العلاقة بين المستقيمين ص = −٣س + ٥ و س − ٣ص = ٦؟
الحل:
  1. ميل الأول −٣.
  2. نعيد كتابة الثاني: ص = ١٣س − ٢ ⟸ ميله ١٣.
  3. حاصل ضرب الميلين −٣ × ١٣ = −١ ⟸ متعامدان.
٧ما حل المتباينة −٩ت ≤ −١٨؟
الحل:
  1. نقسم الطرفين على −٩ ونعكس رمز المتباينة لأننا قسمنا على عدد سالب.
  2. ت ≥ ٢ ، ومجموعة الحل {ت | ت ≥ ٢}
٨ما حل المتباينة −٢ر − ٥ ≥ ٣ر + ١٠؟
الحل:
  1. نطرح ٣ر من الطرفين: −٥ر − ٥ ≥ ١٠
  2. نضيف ٥ إلى الطرفين: −٥ر ≥ ١٥
  3. نقسم على −٥ ونعكس الرمز لأن القاسم سالب: ر ≤ −٣
  4. مجموعة الحل: {ر | ر ≤ −٣}
٩ما حل المتباينة المركبة ن + ٤ ≤ ١ أو ٢ن − ٣ ≥ −١؟
الحل:
  1. نحل كل متباينة على حدة، ثم نأخذ الاتحاد (أي عدد يحقق إحداهما على الأقل).
  2. الأولى: نطرح ٤ ⟸ ن ≤ −٣
  3. الثانية: نضيف ٣ ثم نقسم على ٢ ⟸ ن ≥ ١
  4. الحل: ن ≤ −٣ أو ن ≥ ١
١٠ما حل المتباينة |ر + ٣| ≤ ٥؟
الحل:
  1. |أ| ≤ ب (حيث ب > ٠) تكافئ متباينة «و»: −٥ ≤ أ ≤ ٥
  2. −٥ ≤ ر + ٣ ≤ ٥
  3. نطرح ٣ للأجزاء الثلاثة: −٨ ≤ ر ≤ ٢
  4. المعنى: المسافة بين ر و −٣ أقل من أو تساوي ٥.
